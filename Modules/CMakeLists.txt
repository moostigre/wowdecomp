# Every enabled module supplies a namespaced descriptor accessor. Explicit calls
# in the generated registry avoid static-initializer and archive dead-strip bugs.
function(wow_add_client_module)
    cmake_parse_arguments(MODULE "" "NAME;FACTORY" "SOURCES" ${ARGN})
    if (MODULE_UNPARSED_ARGUMENTS)
        message(FATAL_ERROR "Unknown module arguments: ${MODULE_UNPARSED_ARGUMENTS}")
    endif ()
    if (NOT MODULE_NAME OR NOT MODULE_FACTORY OR NOT MODULE_SOURCES)
        message(FATAL_ERROR "wow_add_client_module requires NAME, FACTORY and SOURCES")
    endif ()
    if (NOT MODULE_FACTORY MATCHES "^[A-Za-z_][A-Za-z0-9_]*$")
        message(FATAL_ERROR "Module FACTORY must be an unqualified C++ identifier")
    endif ()
    get_property(names GLOBAL PROPERTY WOW_CLIENT_MODULE_NAMES)
    if (MODULE_NAME IN_LIST names)
        message(FATAL_ERROR "Duplicate client module name: ${MODULE_NAME}")
    endif ()
    set_property(GLOBAL APPEND PROPERTY WOW_CLIENT_MODULE_NAMES "${MODULE_NAME}")
    set_property(GLOBAL APPEND PROPERTY WOW_CLIENT_MODULE_FACTORIES "${MODULE_FACTORY}")
    foreach (source IN LISTS MODULE_SOURCES)
        get_filename_component(absolute_source "${source}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
        target_sources(ClientModules PRIVATE "${absolute_source}")
    endforeach ()
endfunction()

option(WOW_MODULE_EXAMPLE "Compile the example client module" OFF)
if (WOW_MODULE_EXAMPLE)
    add_subdirectory(Example)
endif ()
if (WOW_MODULE_XSPECS)
    add_subdirectory(XSpecs)
endif ()
# Add future module subdirectories above registry generation.
get_property(factories GLOBAL PROPERTY WOW_CLIENT_MODULE_FACTORIES)
set(registry "#include \"Modules/ModuleHost.h\"\nnamespace ClientModules {\n")
foreach (factory IN LISTS factories)
    string(APPEND registry "const Module& ${factory}();\n")
endforeach ()
string(APPEND registry "bool RegisterBuiltInModules(Host& host, std::string& error) {\nerror.clear();\n(void)host;\n")
foreach (factory IN LISTS factories)
    string(APPEND registry "if (!host.Register(${factory}(), error)) return false;\n")
endforeach ()
string(APPEND registry "return true;\n}\n}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ModuleRegistry.cpp" "${registry}")
target_sources(ClientModules PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/ModuleRegistry.cpp")
